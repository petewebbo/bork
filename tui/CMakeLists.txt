cmake_minimum_required(VERSION 3.16)
project(tui VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CURSES_NEED_NCURSES TRUE)
find_package(Curses REQUIRED)
include(../cmake/SPSCQueue.cmake)
find_package(Threads REQUIRED)
add_executable(tui main.cpp grid_tui.cpp)
target_include_directories(tui PRIVATE ../inc ${CURSES_INCLUDE_DIRS})
target_link_libraries(tui PRIVATE ${CURSES_LIBRARIES} SPSCQueue::SPSCQueue Threads::Threads)
